* 1. Escribe un programa en C que pida dos números enteros y determine si ambos son positivos usando el operador AND.

#+BEGIN_SRC C
#include <stdio.h>
int main() 
{
    int num1, num2, edad, num;

    // Ejercicio 1: Determinar si ambos números son positivos
    printf("Ejercicio 1: Introduzca dos números enteros: ");
    scanf("%d %d", &num1, &num2);
    if (num1 > 0 && num2 > 0) {
        printf("Ambos números son positivos.\n");
    } else {
        printf("Al menos uno de los números no es positivo.\n");
    }
}

#+END_SRC

* 2. Escribe un programa en C que pida la edad de una persona y determine si es un adolescente (entre 13 y 19 años) usando AND.

#+BEGIN_SRC C
#include <stdio.h>
int main() 
{
    int edad;
    printf("\nEjercicio 2: Introduzca la edad de la persona: ");
    scanf("%d", &edad);
    if (edad >= 13 && edad <= 19) 
    {
        printf("La persona es un adolescente.\n");
    } else 
    {
        printf("La persona no es un adolescente.\n");
    }
}

#+END_SRC

* 3. Escribe un programa en C que verifique si un número ingresado es divisible por 3 o por 5 usando OR).

#+BEGIN_SRC C
#include <stdio.h>
int main() 
{
    int num;
    printf("\nEjercicio 3: Introduzca un número entero: ");
    scanf("%d", &num);
    if (num % 3 == 0 || num % 5 == 0) 
    {
        printf("El número es divisible por 3 o por 5.\n");
    } else 
    {
        printf("El número no es divisible por 3 ni por 5.\n");
    }
}

* 4. Escribe un programa en C que valide si un usuario ha ingresado un número dentro del rango de 1 a 100 o si ha ingresado el número 0 usando OR y AND).

#+BEGIN_SRC C
#include <stdio.h>
int main() 
{
    int num;
    printf("\nEjercicio 4: Introduzca un número entero: ");
    scanf("%d", &num);
    if ((num >= 1 && num <= 100) || num == 0) 
    {
        printf("El número está en el rango de 1 a 100 o es 0.\n");
    } else 
    {
        printf("El número no está en el rango de 1 a 100 y no es 0.\n");
    }
}
#+END_SRC
